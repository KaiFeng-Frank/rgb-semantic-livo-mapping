cmake_minimum_required(VERSION 3.16)
project(semantic_frontend_contracts LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(semantic_frontend INTERFACE)
target_include_directories(semantic_frontend INTERFACE include)
target_link_libraries(semantic_frontend INTERFACE Eigen3::Eigen)
target_compile_features(semantic_frontend INTERFACE cxx_std_17)
enable_testing()
add_executable(test_semantic_frontend tests/test_semantic_frontend.cpp)
target_link_libraries(test_semantic_frontend PRIVATE semantic_frontend)
target_compile_options(test_semantic_frontend PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME semantic_frontend_contracts COMMAND test_semantic_frontend)
